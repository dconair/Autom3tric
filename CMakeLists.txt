cmake_minimum_required(VERSION 3.22)
project(AutoDJ VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

if(ANDROID)
  set(_default_tests OFF)
else()
  set(_default_tests ON)
endif()
option(AUTODJ_BUILD_TESTS "Build GoogleTest suites" ${_default_tests})
option(AUTODJ_BUILD_BENCH "Build benchmark / stress / render tools" ${_default_tests})
option(AUTODJ_RT_GUARD "Audio-thread allocation detector (replaces global operator new/delete)" ${_default_tests})
option(AUTODJ_BUILD_JNI "Build the JNI library with the platform audio driver (macOS desktop app; Android sets this up itself)" OFF)
option(AUTODJ_WERROR "Treat warnings as errors" OFF)
option(AUTODJ_FORCE_SCALAR "Disable the NEON/SSE2 grain kernels (A/B testing)" OFF)
set(AUTODJ_SANITIZE "" CACHE STRING "Sanitizers, e.g. address,undefined or thread")

include(cmake/Dependencies.cmake)
include(cmake/Options.cmake)

if(AUTODJ_BUILD_TESTS)
  enable_testing()
endif()

add_subdirectory(native)
